# KONG 商店盲拍（Milestone 3）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 引入**局间商店盲拍**：每局结算展示后（把末除外）进入商店，展示至多 3 件随机遗物（v1 池 2 件），每位存活玩家限选 1 件、输入出价（货币、密封不可见金额，只显示"几人已选"），全员提交/跳过 → 统一揭示 → 每件价高者得（同价先到先得），胜者扣货币获遗物、败者本轮失去资格，全员跳过则直接下一局。

**Architecture:** 全服务器权威。新增 `Phase.SHOP`（值 10）；非把末结算后由现有 `series_timer`（GAME_OVER 停留）触发 `_start_shop()`（替代原来"自动开下一局"）；商店状态 `shop`（offers/bids/skip），密封出价**不进快照**（只投影 offers 的 `bidders` 计数与 `submitted`）；全员提交后 `_resolve_shop()`（价高者得、同价按提交序先到先得），胜者扣货币、`run_state.relics[relic_id]` 覆写（同种替代），随后 `_deal_next_match()` 开下一局。遗物定义放 `scripts/core/relics.gd`（数据 + 池；效果钩子属 Milestone 4，本里程碑购买即入库但无效果）。客户端：商店面板做成 `.tscn` 场景实体（仿 bet_panel：暗遮罩 + 不透明面板、W/S 调价、Enter 确认、跳过）。

**Tech Stack:** Godot 4.6 / GDScript / ENet。headless 单测（`tests/verify_shop.tscn`）。

**Spec:** `docs/superpowers/specs/2026-09-07-series-shop-relics-design.md` §5；用户确认（2026-09-08）：每局结算后进（把末除外）、允许跳过、每人限选 1 件、同价先到先得、出价用货币（影响把末排名）。

## Global Constraints

- 保持 `GameState` 服务器权威、私有快照、贴牌服务器时序三条架构不变量不变。
- 密封出价金额**不得进入任何快照**（只投影计数与提交状态）；`shop_result` 公开（谁以多少拍到）。
- 遗物效果在本里程碑为 no-op（只入库）；`enable_relics` 基础模式不受影响（商店只出现在系列赛局间，基础单局流程不变）。
- `_server_next_match`（GAME_OVER 房主"再来一局"）保留；商店裁决走内部 `_deal_next_match()`（`match_number += 1; _deal_new_match()`）。
- 新 UI 优先 `.tscn`（商店面板 `scenes/ui/shop_panel.tscn`）。
- 改协议前更新 `docs/网络协议_V1.md`；改规则前更新 `docs/KONG_开发文档.md`。
- 每次改动跑回归 `tools/run_all_tests.sh --quick`；提交 `feat/fix/doc` 分类。

```gdscript
## 进入商店阶段：随机展示至多 3 件遗物，等待全员出价或跳过。
func _start_shop() -> void:
	if _series_finished():
		return
	shop = {"offers": _pick_shop_offers(), "bids": {}, "skip": {}, "_next_order": 0}
	shop_result = {}
	phase = Phase.SHOP
	_add_log("进入商店：每人限选 1 件竞拍（密封出价）。")
	_broadcast_state()

func _pick_shop_offers() -> Array:
	var pool: Array = Relics.pool().duplicate()
	pool.shuffle()
	var offers: Array = []
	var count := mini(3, pool.size())
	for i in count:
		var relic: Dictionary = pool[i]
		offers.append({"index": i, "relic_id": str(relic.id), "name": str(relic.name), "min_bid": int(relic.get("min_bid", Relics.RELIC_MIN_BID))})
	return offers
```

新增：

```gdscript
## 商店快照：只投影 offers 的 bidders 计数与已提交名单，绝不含出价金额（密封）。
static func _shop_snapshot(state: Node) -> Dictionary:
	var offers: Array = []
	for offer in state.shop.get("offers", []):
		var count := 0
		for seat in state.shop.get("bids", {}):
			if int(state.shop.bids[seat].offer) == int(offer.index):
				count += 1
		offers.append({"id": str(offer.relic_id), "name": str(offer.name), "min_bid": int(offer.min_bid), "bidders": count})
	var submitted: Array = []
	for seat in state.shop.get("bids", {}):
		submitted.append(int(seat))
	for seat in state.shop.get("skip", {}):
		submitted.append(int(seat))
	return {"offers": offers, "submitted": submitted}
```

```gdscript
	if multiplayer.is_server():

func server_shop_bid(offer: int, amount: int, action_id: String) -> void:
	if multiplayer.is_server():
		_server_shop_bid(_peer_to_seat(multiplayer.get_remote_sender_id()), offer, amount, action_id)

func _server_shop_bid(sender: int, offer: int, amount: int, action_id := "") -> void:
	if phase != Phase.SHOP:
		_reject(sender, RejectCode.INVALID_PHASE, action_id)
		return
	if _guard_suspended(sender, action_id):
		return
	if _guard_spectator(sender, action_id):
		return
	if not players.has(sender) or _shop_submitted(sender):
		_reject(sender, RejectCode.DUPLICATE_OR_EXPIRED_ACTION, action_id)
		return
	var offers: Array = shop.get("offers", [])
	if offer < 0 or offer >= offers.size():
		_reject(sender, RejectCode.INVALID_OFFER, action_id)
		return
	var min_bid: int = int(offers[offer].min_bid)
	if amount < min_bid:
		_reject(sender, RejectCode.INVALID_AMOUNT, action_id)
		return
	if amount > int(players[sender].currency):
		_reject(sender, RejectCode.INVALID_AMOUNT, action_id)
		return
	shop.bids[sender] = {"offer": offer, "amount": amount, "order": int(shop.get("_next_order", 0))}
	shop["_next_order"] = int(shop.get("_next_order", 0)) + 1
	_add_log("%s 对 %s 出价。" % [players[sender].name, offers[offer].name])
	if _shop_all_submitted():
		_resolve_shop()
	else:
		_broadcast_state()

func request_shop_skip(action_id := "") -> void:
	if multiplayer.is_server():
		_server_shop_skip(_peer_to_seat(1), action_id)
	else:
		server_shop_skip.rpc_id(1, action_id)

@rpc("any_peer", "reliable")
func server_shop_skip(action_id: String) -> void:
	if multiplayer.is_server():
		_server_shop_skip(_peer_to_seat(multiplayer.get_remote_sender_id()), action_id)

func _server_shop_skip(sender: int, action_id := "") -> void:
	if phase != Phase.SHOP:
		_reject(sender, RejectCode.INVALID_PHASE, action_id)
		return
	if _guard_suspended(sender, action_id):
		return
	if _guard_spectator(sender, action_id):
		return
	if not players.has(sender) or _shop_submitted(sender):
		_reject(sender, RejectCode.DUPLICATE_OR_EXPIRED_ACTION, action_id)
		return
	shop.skip[sender] = true
	_add_log("%s 跳过商店。" % players[sender].name)
	if _shop_all_submitted():
		_resolve_shop()
	else:
		_broadcast_state()

func _shop_submitted(seat: int) -> bool:
	return shop.get("bids", {}).has(seat) or shop.get("skip", {}).has(seat)

func _shop_all_submitted() -> bool:
	for seat in _alive_order():
		if not _shop_submitted(int(seat)):
			return false
	return true

## 商店裁决：每件 offer 价高者得（同价按提交序先到先得）；胜者扣货币、遗物入库（同 id 覆写）。
func _resolve_shop() -> void:
	var result: Dictionary = {}
	for offer in shop.get("offers", []):
		var oi: int = int(offer.index)
		var best_seat := -1
		var best_amount := -1
		var best_order := 1 << 30
		for seat in shop.get("bids", {}):
			var b: Dictionary = shop.bids[seat]
			if int(b.offer) != oi:
				continue
			var amt: int = int(b.amount)
			var ord: int = int(b.order)
			if amt > best_amount or (amt == best_amount and ord < best_order):
				best_amount = amt
				best_order = ord
				best_seat = int(seat)
		if best_seat >= 0:
			var relic_id: String = str(offer.relic_id)
			players[best_seat].currency = int(players[best_seat].currency) - best_amount
			run_state["relics"][relic_id] = Relics.def_by_id(relic_id)
			_add_log("%s 以 %d 货币拍得 %s。" % [players[best_seat].name, best_amount, offer.name])
			result[str(oi)] = {"relic_id": relic_id, "name": str(offer.name), "winner": best_seat, "amount": best_amount}
	shop_result = result
	shop.clear()
	_deal_next_match()
```

```gdscript
func _unhandled_input(event: InputEvent) -> void:
```
