# 房间池（Room Pool）实现计划

> **For agentic workers:** 本计划按任务顺序执行，每个任务自带验证。项目规则：**不自动 git commit**（由用户决定提交时机），因此每个任务以"验证通过"为完成标志。

**Goal:** VPS 上常驻 4 个 dedicated 实例组成房间池，客户端点"在线房间"即由目录分配空闲实例并拿到房间码，跨网络无需端口转发。

**Architecture:** 扩展 `server/directory/directory.py` 为分配器（房间状态 `idle/reserved/in_use` + `/allocate_room`）；专用服务器加 `--pool` 以 idle 登记、房间结束换码回池；客户端创建入口改为"在线/局域网"二选一，在线走"分配 → 查码 → 直连"。

**Tech Stack:** Godot 4.6 (GDScript, ENet)、Python 3 标准库 HTTP。

## Global Constraints

- 不改对局规则、状态机、快照投影、`HiddenInfo`、既有 RPC。
- 保持 `Network.host_game` / player-host 路径可用（局域网模式）。
- 池大小 4（端口 7007–7010）、预约超时 30s、心跳 10s。
- 所有新增网络调用都走既有 `DirectoryClient`（错误回调 `cb(ok, data)`）。
- **不自动提交 git**；只跑 headless 测试，不启动 GUI。

新增方法：

```python
    def allocate(self, now=None):
        now = time.time() if now is None else now
        with self._lock:
            self.sweep(now)
            idle = [c for c, r in self._rooms.items() if r["state"] == IDLE]
            if not idle:
                return None
            idle.sort(key=lambda c: self._rooms[c]["last_alloc"])
            code = idle[0]
            r = self._rooms[code]
            r["state"] = RESERVED
            r["reserved_at"] = now
            r["last_alloc"] = now
            return {"code": code, "host": r["host"], "port": r["port"]}

    def stats(self):
        with self._lock:
            idle = sum(1 for r in self._rooms.values() if r["state"] == IDLE)
            busy = sum(1 for r in self._rooms.values() if r["state"] != IDLE)
            return {"rooms": len(self._rooms), "idle": idle, "busy": busy}
```

`join()` 改为：命中后若 `r["state"] == IDLE` 返回 `None`；否则置 `IN_USE`、`reserved_at=0` 再返回地址。

```gdscript
class_name PoolRegistrar
extends RefCounted
## 专用服务器在房间池中的登记/轮换。目录调用注入，便于 headless 单测。

const STATE_IDLE := "idle"
const STATE_IN_USE := "in_use"

var code := ""
var state := STATE_IDLE
var last_error := ""

var _create: Callable
var _close: Callable

func setup(create_cb: Callable, close_cb: Callable) -> void:
	_create = create_cb
	_close = close_cb

func register_idle(port: int, mode: String, max_players: int, cb := Callable()) -> void:
	_register(port, mode, max_players, STATE_IDLE, cb)

func re_register(port: int, mode: String, max_players: int, in_use: bool, cb := Callable()) -> void:
	_register(port, mode, max_players, STATE_IN_USE if in_use else STATE_IDLE, cb)

func rotate_after_room(port: int, mode: String, max_players: int, cb := Callable()) -> void:
	if not code.is_empty() and _close.is_valid():
		_close.call(code, Callable())
	code = ""
	register_idle(port, mode, max_players, cb)

func _register(port: int, mode: String, max_players: int, new_state: String, cb: Callable) -> void:
	state = new_state
	code = ""
	if not _create.is_valid():
		last_error = "NO_CLIENT"
		if cb.is_valid():
			cb.call(false, {"error": last_error})
		return
	_create.call(port, mode, max_players, new_state, func(ok: bool, data: Dictionary):
		if ok and data.has("code"):
			code = str(data["code"])
			last_error = ""
		else:
			last_error = str(data.get("error", "?"))
		if cb.is_valid():
			cb.call(ok, data))
```

```gdscript
var _pool: PoolRegistrar = null
var _pool_port := 0

func _start_pool(url: String, port: int, advertise: String) -> void:
	if url.is_empty():
		print("[KONG] pool mode needs --directory; players must join by IP")
		return
	_pool_port = port
	var dc := DirectoryClient.new()
	dc.setup(url)
	add_child(dc)
	_pool = PoolRegistrar.new()
	_pool.setup(
		func(p: int, mode: String, max_players: int, state: String, cb: Callable):
			dc.create_room(p, mode, max_players, cb, advertise, "", state),
		func(c: String, cb: Callable):
			dc.close_room(c, cb))
	_pool.register_idle(port, "dedicated", KongRules.MAX_PLAYERS, func(ok: bool, _d: Dictionary):
		if ok:
			print("[KONG] pool room code: %s" % _pool.code)
		else:
			print("[KONG] pool register failed: %s" % _pool.last_error))
	GameState.room_ended.connect(func(): _on_pool_room_ended(advertise, dc))
	var t := Timer.new()
	t.wait_time = 10.0
	t.timeout.connect(func(): _on_pool_heartbeat(dc))
	add_child(t)
	t.start()

func _on_pool_room_ended(advertise: String, dc: DirectoryClient) -> void:
	if _pool == null:
		return
	print("[KONG] room ended; recycling pool instance on UDP %d" % _pool_port)
	_pool.rotate_after_room(_pool_port, "dedicated", KongRules.MAX_PLAYERS, func(ok: bool, _d: Dictionary):
		if ok:
			print("[KONG] new room code: %s" % _pool.code)
		else:
			print("[KONG] pool re-register failed: %s" % _pool.last_error))

func _on_pool_heartbeat(dc: DirectoryClient) -> void:
	if _pool == null or _pool.code.is_empty():
		return
	dc.heartbeat(_pool.code, func(ok: bool, _d: Dictionary):
		if not ok:
			_pool.re_register(_pool_port, "dedicated", KongRules.MAX_PLAYERS,
				not GameState.players.is_empty(), func(_ok2: bool, _d2: Dictionary):
					print("[KONG] pool re-registered code: %s" % _pool.code)))
```

```gdscript
extends Node
## 房间池端到端（需先启动目录与一个 --pool 实例）：
##   python3 server/directory/directory.py --port 18191
##   Godot --headless --path . res://scenes/server.tscn -- --port 17107 --pool \
##     --directory http://127.0.0.1:18191
##   Godot ... res://tests/verify_room_pool_e2e.tscn -- -role creator -dir http://127.0.0.1:18191
##   Godot ... res://tests/verify_room_pool_e2e.tscn -- -role joiner  -dir http://127.0.0.1:18191
## creator 把码写入 user://pool_code.txt，joiner 读取后加入。

var role := ""
var dir_url := "http://127.0.0.1:18191"
var deadline := 45.0
var done := false
const CODE_FILE := "user://pool_code.txt"

func _ready() -> void:
	_parse_args()
	if role.is_empty():
		_fail("need -role creator|joiner")
		return
	GameState.lobby_updated.connect(_on_lobby)
	if role == "creator":
		DirAccess.remove_absolute(ProjectSettings.globalize_path(CODE_FILE))
		_create()
	else:
		_poll_code()

func _parse_args() -> void:
	var args := OS.get_cmdline_user_args()
	var i := 0
	while i < args.size():
		match args[i]:
			"-role":
				i += 1; role = args[i] if i < args.size() else ""
			"-dir":
				i += 1; dir_url = args[i] if i < args.size() else dir_url
		i += 1

func _process(delta: float) -> void:
	if done:
		return
	deadline -= delta
	if deadline <= 0.0:
		_fail("timeout")

func _create() -> void:
	var dc := DirectoryClient.new()
	dc.setup(dir_url)
	add_child(dc)
	dc.allocate_room(func(ok: bool, data: Dictionary):
		if not ok:
			_fail("allocate failed: %s" % str(data.get("error", "?")))
			return
		var code := str(data.get("code", ""))
		dc.join_room(code, func(ok2: bool, data2: Dictionary):
			if not ok2 or not data2.has("host"):
				_fail("join_room failed")
				return
			_write_file(CODE_FILE, code)
			print("[creator] code=%s → connect %s:%d" % [code, str(data2["host"]), int(data2["port"])])
			Network.join_game(str(data2["host"]), {"name": "创建者A"}, int(data2["port"]))))

func _poll_code() -> void:
	var waited := 0.0
	while waited < deadline:
		if FileAccess.file_exists(CODE_FILE):
			var fa := FileAccess.open(CODE_FILE, FileAccess.READ)
			if fa != null:
				var c := fa.get_as_text().strip_edges()
				fa.close()
				if not c.is_empty():
					_join(c)
					return
		await get_tree().create_timer(0.3).timeout
		waited += 0.3
	_fail("no code file")

func _join(c: String) -> void:
	var dc := DirectoryClient.new()
	dc.setup(dir_url)
	add_child(dc)
	dc.join_room(c, func(ok: bool, data: Dictionary):
		if ok and data.has("host"):
			Network.join_game(str(data["host"]), {"name": "朋友B"}, int(data["port"]))
		else:
			_fail("join_room failed: %s" % str(data.get("error", "?"))))

func _on_lobby(lobby: Dictionary) -> void:
	var players: Array = lobby.get("players", [])
	var host_id := int(lobby.get("room_host", -1))
	var your_id := int(lobby.get("your_id", -1))
	if role == "creator" and players.size() >= 1:
		if host_id == your_id:
			_pass("创建者成为房主（%d 人）" % players.size())
		else:
			_fail("创建者不是房主")
	elif role == "joiner" and players.size() >= 2:
		_pass("朋友凭码加入成功（%d 人）" % players.size())

func _write_file(path: String, text: String) -> void:
	var fa := FileAccess.open(path, FileAccess.WRITE)
	if fa != null:
		fa.store_string(text)
		fa.close()

func _pass(msg: String) -> void:
	done = true
	print("VERIFY %s PASS: %s" % [role, msg])
	await get_tree().create_timer(1.0).timeout
	get_tree().quit(0)

func _fail(msg: String) -> void:
	if done:
		return
	done = true
	print("VERIFY %s FAIL: %s" % [role, msg])
	get_tree().quit(1)
```
